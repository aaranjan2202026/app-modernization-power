```mermaid
    alt Validation Failed
        Controller-->>Client: 400 Bad Request
    else Validation Passed
        Controller->>Service: ProcessRequest(data)
        activate Service
        Note over Service: Business Logic<br/>Rule Validation
        
        Service->>Repository: GetRelatedData(id)
        activate Repository
        Repository->>Database: SELECT Query
        activate Database
        Database-->>Repository: Result Set
        deactivate Database
        Repository-->>Service: Entity Data
        deactivate Repository
        
        Service->>External: CallExternalAPI(data)
        activate External
        External-->>Service: API Response
        deactivate External
        
        Service->>Repository: Save(entity)
        activate Repository
        Repository->>Database: INSERT/UPDATE
        activate Database
        Database-->>Repository: Success
        deactivate Database
        Repository-->>Service: Saved Entity
        deactivate Repository
        
        Service-->>Controller: Result
        deactivate Service
        
        Controller-->>Client: 200 OK with Response
    end
    deactivate Controller
```
